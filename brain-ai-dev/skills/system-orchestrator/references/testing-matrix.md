# Testing Matrix

## Overview

The orchestrator test suite verifies correctness across flag combinations,
modality sets, devices, and dependency configurations. Tests are parameterized
to achieve coverage without exponential blowup.

## Test Categories

### 1. Preset Instantiation Matrix

For each `BrainAIConfig` preset, verify instantiation succeeds with and without
optional dependencies:

| Preset | Deps Present | Deps Missing | Expected |
|--------|-------------|--------------|----------|
| `minimal()` | all | all | OK (always) |
| `for_vision_only()` | all | all | OK |
| `production_1b()` | all | all | OK |
| `production_3b()` | all | ncps | OK (GRU fallback) |
| `production_7b()` | all | htm_core | OK (LSTM fallback) |
| `production_7b()` | all | learn2learn | OK (custom MAML) |
| `production_7b()` | all | pymdp | OK (continuous AIF) |

**Test pattern**:

```python
@pytest.mark.parametrize("preset", [
    "minimal", "for_vision_only", "production_1b", "production_3b", "production_7b"
])
@pytest.mark.parametrize("missing_dep", [None, "ncps", "htm.core", "learn2learn", "pymdp"])
def test_preset_instantiation(preset, missing_dep, monkeypatch):
    if missing_dep:
        monkeypatch.setattr(f"brain_ai.system.{missing_dep}", None)
        # or: monkeypatch importlib to raise ImportError

    config = getattr(BrainAIConfig, preset)()
    brain = BrainAI(config)

    # Verify no ImportError leaked
    assert brain is not None

    # Verify fallback was recorded
    report = brain.deps_report()
    if missing_dep:
        assert "missing" in report.get(missing_dep.replace(".", "_"), "")
```

### 2. Contract Tests

For each modality set, verify forward output contracts:

```python
MODALITY_COMBOS = [
    ["vision"],
    ["text"],
    ["audio"],
    ["sensors"],
    ["vision", "text"],
    ["vision", "text", "audio"],
    ["vision", "text", "audio", "sensors"],
]

@pytest.mark.parametrize("modalities", MODALITY_COMBOS)
def test_forward_contract(modalities):
    config = BrainAIConfig.minimal()
    config.modalities = modalities
    brain = BrainAI(config)

    inputs = make_sample_inputs(modalities, batch_size=2, device="cpu")
    output = brain(inputs, return_details=True)

    # Shape checks
    assert output.output.shape[0] == 2  # Batch dim
    assert output.confidence.shape == (2, 1)

    # Device checks
    assert output.output.device.type == "cpu"

    # Details schema stability
    assert isinstance(output.details, SystemDetails)
    # Fields may be None but must exist
    assert hasattr(output.details, "encoder")
    assert hasattr(output.details, "workspace")
    assert hasattr(output.details, "htm")
    assert hasattr(output.details, "reasoning")
    assert hasattr(output.details, "meta")
    assert hasattr(output.details, "engram")
```

### 3. Flag Bypass Invariants

For each feature flag, verify disabling produces valid output:

```python
FLAGS = ["use_snn", "use_htm", "use_workspace", "use_symbolic", "use_meta", "use_engram"]

@pytest.mark.parametrize("disabled_flag", FLAGS)
def test_flag_bypass(disabled_flag):
    config = BrainAIConfig.minimal()
    config.modalities = ["vision"]

    # Enable all flags first
    for flag in FLAGS:
        setattr(config, flag, True)

    # Then disable the target flag
    setattr(config, disabled_flag, False)

    brain = BrainAI(config)
    inputs = make_sample_inputs(["vision"], batch_size=2)

    # With flag disabled
    out_disabled = brain(inputs, return_details=True)

    # Schema must be identical to all-enabled
    config_all = BrainAIConfig.minimal()
    for flag in FLAGS:
        setattr(config_all, flag, True)
    brain_all = BrainAI(config_all)
    out_all = brain_all(inputs, return_details=True)

    # Same output fields
    assert type(out_disabled) == type(out_all)
    assert out_disabled.output.shape == out_all.output.shape
    assert out_disabled.confidence.shape == out_all.confidence.shape

    # Details keys same
    if out_disabled.details and out_all.details:
        for field_name in ["encoder", "workspace", "htm", "reasoning", "meta", "engram"]:
            d = getattr(out_disabled.details, field_name, None)
            a = getattr(out_all.details, field_name, None)
            if a is not None:
                assert d is not None or d is None  # Can be None but not missing type
```

### 4. Pairwise Flag Combination Tests

Full powerset of 6 flags = 64 combinations. Use pairwise coverage for practical
test count (~15-20 combinations):

```python
# Generated by scripts/gen_flag_tests.py
PAIRWISE_COMBOS = [
    {"use_snn": True,  "use_htm": True,  "use_workspace": True,  "use_symbolic": True,  "use_meta": True,  "use_engram": True},
    {"use_snn": False, "use_htm": False, "use_workspace": False, "use_symbolic": False, "use_meta": False, "use_engram": False},
    {"use_snn": True,  "use_htm": False, "use_workspace": True,  "use_symbolic": False, "use_meta": True,  "use_engram": False},
    {"use_snn": False, "use_htm": True,  "use_workspace": False, "use_symbolic": True,  "use_meta": False, "use_engram": True},
    # ... ~16 more rows covering all pairwise interactions
]

@pytest.mark.parametrize("flags", PAIRWISE_COMBOS)
def test_pairwise_flags(flags):
    config = BrainAIConfig.minimal()
    config.modalities = ["vision"]
    for flag, value in flags.items():
        setattr(config, flag, value)

    brain = BrainAI(config)
    inputs = make_sample_inputs(["vision"], batch_size=2)
    output = brain(inputs, return_details=True)

    assert output.output is not None
    assert not torch.isnan(output.output).any()
    assert output.confidence is not None
```

### 5. Determinism Test

```python
def test_determinism():
    config = BrainAIConfig.minimal()
    brain = BrainAI(config)
    inputs = make_sample_inputs(["vision"], batch_size=2)

    set_global_seed(42)
    out1 = brain(inputs, return_details=True)

    set_global_seed(42)
    out2 = brain(inputs, return_details=True)

    assert torch.equal(out1.output, out2.output)
    assert torch.equal(out1.confidence, out2.confidence)

    if out1.details.workspace and "winners" in out1.details.workspace:
        assert torch.equal(
            out1.details.workspace["winners"],
            out2.details.workspace["winners"]
        )
```

### 6. Device Tests (CPU + CUDA)

```python
@pytest.mark.parametrize("device", ["cpu", pytest.param("cuda", marks=pytest.mark.skipif(
    not torch.cuda.is_available(), reason="CUDA not available"))])
def test_device(device):
    config = BrainAIConfig.minimal()
    brain = BrainAI(config).to(device)
    inputs = make_sample_inputs(["vision"], batch_size=2, device=device)
    output = brain(inputs)

    assert output.device == torch.device(device)
    # No device mismatches
    for name, param in brain.named_parameters():
        assert param.device == torch.device(device), f"{name} on wrong device"
```

### 7. return_details Consistency

```python
def test_return_details_consistency():
    config = BrainAIConfig.minimal()
    brain = BrainAI(config)
    inputs = make_sample_inputs(["vision"], batch_size=2)

    out_simple = brain(inputs, return_details=False)
    out_detailed = brain(inputs, return_details=True)

    # Main output should be identical
    assert torch.allclose(out_simple.output, out_detailed.output, atol=1e-6)

    # Details present only when requested
    assert out_simple.details is None
    assert out_detailed.details is not None
```

### 8. Streaming State Test

```python
def test_streaming_state():
    config = BrainAIConfig.minimal()
    brain = BrainAI(config)

    state = BrainAIState()
    outputs = []

    for t in range(5):
        inputs = make_sample_inputs(["vision"], batch_size=1)
        output, state = brain.step(inputs, state)
        outputs.append(output)

    # State should accumulate
    assert state.step_count == 5

    # Outputs should be valid at each step
    for out in outputs:
        assert not torch.isnan(out.output).any()
```

### 9. AMP Smoke Test

```python
@pytest.mark.skipif(not torch.cuda.is_available(), reason="CUDA required for AMP")
def test_amp_forward():
    config = BrainAIConfig.minimal()
    config.training.use_amp = True
    brain = BrainAI(config).cuda()
    inputs = make_sample_inputs(["vision"], batch_size=2, device="cuda")

    with torch.cuda.amp.autocast():
        output = brain(inputs)

    assert not torch.isnan(output.output).any()
    assert not torch.isinf(output.output).any()
```

## "Done When" Checklist

All of the following must pass before the orchestrator refactor is considered complete:

- [ ] All 5 presets instantiate with all optional deps missing
- [ ] Forward pass succeeds for every modality combination (7 combos)
- [ ] Every flag bypass produces schema-identical output
- [ ] Pairwise flag combinations pass (~20 combos)
- [ ] Determinism test passes with fixed seed
- [ ] CPU and CUDA devices work
- [ ] return_details True/False produces consistent output
- [ ] Streaming state carries forward correctly
- [ ] AMP forward produces no NaN/Inf on CUDA
- [ ] deps_report accurately reflects installed packages
- [ ] Run manifest captures all reproducibility info
- [ ] No device mismatch warnings
- [ ] No silent dtype drift
